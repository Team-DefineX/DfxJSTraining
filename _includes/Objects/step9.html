Neden Prototype’e İhtiyacımız Var ?

Javascript programımızda Car nesnesine ihtiyacımız olduğunu düşünelim. 
    
    let car = {}
    car.brand = 'Volkswagen'
    car.hasElectricModels = true
    car.headquartered = 'Wolfsburg'
    car.numberOfEmployee = 5400

    car.getheadquarteredInfo = function() {
        console.log(`${this.brand} headquartered is ${this.headquartered}`)
    }

    car.getNumberOfEmployeeInfo = function() {
        console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
    }

Bu ihtiyacımızı görür fakat birden çok car nesnesine ihtiyacımız olduğu senaryoyu düşünelim. Muhtemelen bu işi yapması için bir fonksiyon yazardık. 

    
    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
        let car = {}
        car.brand = brand
        car.hasElectricModels = hasElectricModels
        car.headquartered = headquartered
        car.numberOfEmployee = numberOfEmployee

        car.getheadquarteredInfo = function() {
            console.log(`${this.brand} headquartered is ${this.headquartered}`)
        }

        car.getNumberOfEmployeeInfo = function() {
            console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
        }

        return car
    }

    var car = Car('Volkswagen', true, 'Wolfsburg', 5400)

    console.log(car)

Output

              },

                  console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
              },
        hasElectricModels: true,
        headquartered: "Wolfsburg",

Bu güzel ama bunu yapmak için doğru yol bu mu?


Eğer milyonlarca car nesnesine ihtiyacınız olursa bu fonksiyonu kullanmak iyi bir fikir değil çünkü her nesne için hafızada getheadquarteredInfo() ve getNumberOfEmployeeInfo() fonksiyonları oluşacak bu da hafızayı gereksiz yere harcamaktan başka bir şey değil.


Bunun yerine car fonksiyonunda yazdığımız fonksiyonları bir kez oluşturup fonksiyonun içerisinde bu fonksiyonlara referans verirsek getheadquarteredInfo() ve getNumberOfEmployeeInfo() fonksiyonları hafızada her seferinde yeniden oluşmak yerine bir kere oluşur. 

    
    const CarMethods = {
        getheadquarteredInfo : function() {
            console.log(`${this.brand} headquartered is ${this.headquartered}`)
        },
        getNumberOfEmployeeInfo : function() {
            console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
        }
    }

    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
        let car = {}
        car.brand = brand
        car.hasElectricModels = hasElectricModels
        car.headquartered = headquartered
        car.numberOfEmployee = numberOfEmployee

        car.getheadquarteredInfo = CarMethods.getheadquarteredInfo
        car.getNumberOfEmployeeInfo = CarMethods.getNumberOfEmployeeInfo

        return car
    }

    var car = Car('Volkswagen', true, 'Wolfsburg', 5400)

    console.log(car)

Ancak bu yöntemin de bazı dezavantajları vardır. Ne olduğunu tahmin edebilir misiniz? Ben açıklamadan önce bir düşünün.


Diyelim ki daha sonradan Car nesnesine bir fonksiyon ekleme ihtiyacı duydunuz. Bu durumda, önce CarMethods‘a fonksiyonumuzu ekler sonrada Car fonksiyonundan tekrar referansını alırdık. 


    
    const CarMethods = {
        getheadquarteredInfo : function() {
            console.log(`${this.brand} headquartered is ${this.headquartered}`)
        },
        getNumberOfEmployeeInfo : function() {
            console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
        },
        getCompanyInfo : function() {
            console.log(`${this.brand} headquartered is ${this.headquartered} and has ${this.numberOfEmployee} employee`)
        }
    }

    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
        let car = {}
        car.brand = brand
        car.hasElectricModels = hasElectricModels
        car.headquartered = headquartered
        car.numberOfEmployee = numberOfEmployee

        car.getheadquarteredInfo = CarMethods.getheadquarteredInfo
        car.getNumberOfEmployeeInfo = CarMethods.getNumberOfEmployeeInfo
        car.getCompanyInfo = CarMethods.getCompanyInfo

        return car
    }

    var car = Car('Volkswagen', true, 'Wolfsburg', 5400)

    console.log(car)

Bu büyük bir sorun değil fakat bu gibi değişiklikleri yapabileceğimiz ve bu değişikliklerin tüm instance lara yansıtılacağı ortak bir yerin olması elbette güzel olurdu. İşte Javascriptte Prototypeların devreye girdiği yer burasıdır.


Prototype

Prototype temel olarak javascriptte bir fonksiyonun özelliği(Property) dir. Javascriptte bir fonksiyon oluşturduğumuzda javascript motoru fonksiyona otomatik olarak Prototype adında bir özellik(Property) ekler. Bu prototype özelliği bir nesnedir ve bu nesne de constructor adında property e sahiptir. Bu constructor propertysi oluşturulan fonksiyona işaret eder aşağıdaki resimde görüldüğü gibi.

Prototype özelliği ise bahsettiğimiz prototip nesnesine işaret eder.


Fonksiyonun prototip özelliğine functionName.prototype sözdizimini kullanarak erişebiliriz. Ve bir fonksiyonun Prototype nesnesinden, functionName.prototype.constructor ile fonksiyon nesnesine tekrar erişebiliriz. Hala kafanız karıştıysa bunu anlamak için birkaç kod yazalım.


Constructor Fonksiyonları Kullanarak Nesnelere Oluşturmak

Şimdi name ve age olmak üzere iki adet parametre alan constructor fonksiyonumuzu ve bu fonksiyonun prototype nesnesini konsola yazdıralım. 

    
    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
        this.brand = brand
        this.hasElectricModels = brand
        this.headquartered = headquartered
        this.numberOfEmployee = numberOfEmployee
    }

    console.log(Car.prototype)

Aşağıdaki gibi bir çıktı göreceksiniz.


Görüldüğü gibi console.log(Car.prototype) ile fonksiyonumuzun prototype nesnesine ulaştık ve bu nesnenin söylediğim gibi bir constructor özelliği bulunuyor. Şimdilik __proto__ özelliğini kafanıza takmayın. Şimdi ise konsolumuza console.log(Car.prototype.constructor)'ı yazdıralım.



Bir fonksiyon oluşturduğumuzda javascript motoru fonksiyona prototype adında bir property oluşturur bu property ile fonksiyonun prototip nesnesine ulaşırız bu nesnenin içerisinde de constructor adında bir property daha vardır bu property ise fonksiyona işaret eder. şimdi ise bu anlattıklarımızı kanıtlayalım. 

    
    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
        this.brand = brand
        this.hasElectricModels = brand
        this.headquartered = headquartered
        this.numberOfEmployee = numberOfEmployee
    }

    console.log(Car.prototype.constructor === Car)
    console.log(Car.prototype === Car.prototype.constructor.prototype)

Burada ilk logumuzda beklediğimiz sonuç Car fonksiyonumuzun prototype nesnesinde bulunan constructor property sinin bizi tekrar Car fonksiyonuna getirmesi, ikinci logda ise Car fonksiyonunun prototype nesnesindeki constructor propertysi fonksiyonumuza geri işaret ediyor o zaman constructorun prototype'ı Car.prototype'a eşit olmalı.


Output 

    
    true
    true

Yeni bir Car nesnesi oluşturup onu konsola loglayalım.



Peki ya metodlarımız nerde? Onları prototype nesnesine eklemiştik. Ancak car nesnemizde görünmüyor fakat orada başka birşey var __proto__ ne olduğuna bakalım.



Metodlarımız burada. Görünen o ki __proto__ özelliği prototype nesnesine işaret ediyor. Javascirpt, bir instance oluşturduğumzda ona __proto__ adında prototype nesnesine işaret eden bir özellik ekler.


Eğer konunun başından beri anlattığımız şey doğruysa yani __proto__ özelliği prototype nesnesine işaret ediyorsa onu genişlettiğimizde metotları görüyor olmalıyız. 

    
    function Car(brand, hasElectricModels, headquartered, numberOfEmployee) {
         this.brand = brand
         this.hasElectricModels = brand
         this.headquartered = headquartered
         this.numberOfEmployee = numberOfEmployee
     }
 
     Car.prototype.getCompanyInfo = function() {
         console.log(`${this.brand} headquartered is ${this.headquartered} and has ${this.numberOfEmployee} employee`)
     }
 
     Car.prototype.getheadquarteredInfo = function() {
         console.log(`${this.brand} headquartered is ${this.headquartered}`)
     }
 
     Car.prototype.getNumberOfEmployeeInfo = function() {
         console.log(`${this.brand} has ${this.numberOfEmployee} employee`)
     }
 
     const car = new Car('Volkswagen', true, 'Wolfsburg', 5400)
 
     console.log(Car.prototype === car.__proto__)

Output 

    
    true

Son olarak Prototype konusunu özetlmek gerekirse;


Bir fonksiyon oluşturduğumuzda javascript motoru otomatik olarak o fonksiyon için bir Prototype nesnesi oluşturur ve fonksiyona o nesneye erişmek için kullanabilecek bir prototype özelliği ekler ve prototype nesnesi içerisine de constructor adında bir özellik ekler bu da fonksiyona işaret eder.


Constructor fonksiyondan bir instance oluşturduğumuzda ise bu instance __proto__ adında bir özellik eklenir ve bu özellik fonksiyonun prototype nesnesine işaret eder.
