Next Sibling:
        
        <ul id="myList">
            <li>Elma </li>
            <li>Armut </li>
            <li>Portakal </li>
        </ul>
        <script>
            const lastListItem = document.querySelector('li');  // Select first item
            const nextSibling = lastListItem.nextSibling;    // Get next item
            console.log(nextSibling.innerText); // Result: "Armut"
        </script>
Previous Sibling:
    
        <script>
            const lastListItem = document.querySelector('li:last-child');  // Select last item
            const previousSibling = lastListItem.previousSibling;    // get Previous item
            console.log(previousSibling.innerText); // Result: "Armut"
        </script>
Children:
    
        <script>
            const myList = document.getElementById('myList');
            const children = myList.children;
            for (let i = 0; i < children.length; i++) {
              console.log(children[i].innerText);
            }
        </script>
Parent:
    
        <p>
            <span>Bu bir iç içe geçmiş örnek</span>
        </p>
        <script>
            const innerSpan = document.querySelector('span');
            const parentElement = innerSpan.parentNode;
            console.log(parentElement.tagName); // Çıktı: "P"
