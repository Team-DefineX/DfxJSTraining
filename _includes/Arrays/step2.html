Array Methods

JavaScript Array Methods and Properties

Here are some common use cases and operations involving arrays in JavaScript

push(): Adds one or more elements to the end of an array and returns the new length.
pop(): Removes the last element from an array and returns that element.
shift(): Removes the first element from an array and returns that element.
unshift(): Adds one or more elements to the beginning of an array and returns the new length.
concat(): Combines two or more arrays and returns a new array.
slice(): Returns a shallow copy of a portion of an array into a new array.
splice(): Changes the content of an array by adding, removing, or replacing elements.
indexOf(): Returns the first index at which a specified element is found in an array, or -1 if not found.
lastIndexOf(): Returns the last index at which a specified element is found in an array, or -1 if not found.
forEach(): Executes a provided function once for each element in an array.
map(): Creates a new array with the results of calling a provided function on every element in an array.
filter(): Creates a new array with all elements that pass a test implemented by a provided function.
reduce(): Applies a function to an accumulator and each element in an array, resulting in a single value.
reduceRight(): Applies a function to an accumulator and each element in an array (from right to left).
find(): Returns the first element in an array that satisfies a provided testing function.
findIndex(): Returns the index of the first element in an array that satisfies a provided testing function.
some():Tests whether at least one element in an array passes a provided testing function.
every(): Tests whether all elements in an array pass a provided testing function.
sort(): Sorts the elements of an array in place and returns the sorted array.
reverse(): Reverses the order of the elements in an array.
join(): Joins all elements of an array into a string.
toString(): Returns a string representing the specified array and its elements.
isArray(): Determines whether the passed value is an array.

        {

          age: 35,
          isHero: true

        {

          secretIdentity: "Bruce Wayne",
          powers: ["Intelligence", "Combat Skills"],
          age: 40,
          isHero: true
        },
        {
          name: "Wonder Woman",
          secretIdentity: "Diana Prince",
          powers: ["Superhuman Strength", "Lasso of Truth"],
          age: 500,
          isHero: true
        }
    ];
Lets loop throw each item in the array and write to console with basic for loop.
    
    // forEach()
    heroProfiles.forEach(hero => {
      console.log("forEach", hero.name);
    });
Lets map each heroProfile names to new variable called heroNames
    
    // map()
    let heroNames = heroProfiles.map(hero => hero.name);
    console.log("map", heroNames);
Lets map each heroProfile names to new variable called heroNames
    
    // map()
    let heroNames = heroProfiles.map(hero => hero.name);
    console.log("map", heroNames);
Lets filter each heroProfile array to new variable called heroesWithFlight where hero has a special power as Flight
    
    // filter()
    let heroesWithFlight = heroProfiles.filter(hero => hero.powers.includes("Flight"));
    console.log("filter", heroesWithFlight);
Lets reduce total age of each heroProfile and display the sum.
    
    // reduce()
    let totalAge = heroProfiles.reduce((acc, hero) => acc + hero.age, 0);
    console.log("reduce", totalAge);
Lets find hero with name "Batman"
    
    // find()
    let batman = heroProfiles.find(hero => hero.name === "Batman");
    console.log("find",batman);
Lets find hero with power "Super Strength" with some() method
    
    // some()
    let hasSuperStrength = heroProfiles.some(hero => hero.powers.includes("Super Strength"));
    console.log("some",hasSuperStrength);
Lets find only heroes returning true with every() method
    
    // every()
    let allHeroesAreSuper = heroProfiles.every(hero => hero.isHero);
    console.log("every",allHeroesAreSuper);
Lets sort all heroes with age
    
    // sort()
    let sortedHeroes = heroProfiles.sort((a, b) => a.age - b.age);
    console.log("sort", sortedHeroes);
Lets reverse the heroes
    
    // reverse()
    let reversedHeroes = heroProfiles.reverse();
    console.log("reverse", reversedHeroes);
Lets join all hero names with ',' sperated string
    
    // join()
    let heroNamesString = heroProfiles.map(hero => hero.name).join(", ");
    console.log("join", heroNamesString);
Lets convert all array to string
    
    // toString()
    let heroProfilesString = heroProfiles.toString();
    console.log("toString", heroProfilesString);
Lets check if the variable is array
    
    // isArray()
    let isArray = Array.isArray(heroProfiles);
    console.log("isArray", isArray);
Lets find spesific index at the array
    
    // indexOf()
    let index = heroProfiles.indexOf("Batman");
    console.log("indexOf", index);
Lets find last index at the array
    
    // lastIndexOf()
    let lastIndex = heroProfiles.lastIndexOf("Superman");
    console.log("lastIndexOf", lastIndex);
Lets reduce to right side of elements of array
    
    // reduceRight()
    let reversedNames = heroProfiles.reduceRight((acc, hero) => acc + hero.name + " ", "");
    console.log("reduceRight",reversedNames);
Lets find array elements that includes "Superman"
    
    // includes()
    let hasSuperman = heroProfiles.includes("Superman");
    console.log("includes", hasSuperman);
Lets shift elements
    
    // shift()

    console.log("shift", shiftedHero);
Lets unshift elements
    
    // unshift()
    let newLength = heroProfiles.unshift(shiftedHero);
