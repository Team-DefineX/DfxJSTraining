Adding Elements to Array and Calling Array Length

You might need to add new array element to your array.
Lets asume you want to add "Aquaman" and "Catwoman" with all properties to your heroProfile array.
You can simply use push() method to insert new heroes to your heroPorfile array. 

        
        let heroProfiles = [
            {
              name: "Superman",
              secretIdentity: "Clark Kent",
              powers: ["Flight", "Super Strength"],
              age: 35,
              isHero: true
            },
            {
              name: "Batman",
              secretIdentity: "Bruce Wayne",
              powers: ["Intelligence", "Combat Skills"],
              age: 40,
              isHero: true
            },
            {
              name: "Wonder Woman",
              secretIdentity: "Diana Prince",
              powers: ["Superhuman Strength", "Lasso of Truth"],
              age: 500,
              isHero: true
            }
        ];
        
        console.log(heroProfiles.length) // Output: 
        heroProfiles.push({
            name: "Aquaman",
            secretIdentity: "Arthur Curry",
            powers: ["Superhuman Strength", "Marine Telepathy"],
            age: 30,
            isHero: true
         })
         
        console.log(heroProfiles.length) // Output: 4	
        
        let Catwoman = {
            name: "Catwoman",
            secretIdentity: "Selina Kyle",
            powers: ["Agility", "Expert Thief"],
            age: 35,
            isHero: false
        }
        
        heroProfiles.push(Catwoman);
        console.log(heroProfiles.length) // Output: 5
