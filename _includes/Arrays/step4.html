In this example how we can remove element from an Array

You might need to remove element from array.
Lets assume you want to remove "Wonder Woman" from your heroProfiles array.
As a first step we need to find index number of "Wonder Woman" in the array.
After we can use splice() method to remove found index from the array while looping it. 



  let heroProfiles = [
      {
        name: "Superman",
        secretIdentity: "Clark Kent",
        powers: ["Flight", "Super Strength"],
        age: 35,
        isHero: true
      },
      {
        name: "Batman",
        secretIdentity: "Bruce Wayne",
        powers: ["Intelligence", "Combat Skills"],
        age: 40,
        isHero: true
      },
      {
        name: "Wonder Woman",
        secretIdentity: "Diana Prince",
        powers: ["Superhuman Strength", "Lasso of Truth"],
        age: 500,
        isHero: true
      }
  
  // Find the index of Wonder Woman in the array
  let index = heroProfiles.findIndex(hero => hero.name === "Wonder Woman");
              
  // Remove the element at the found index
  if (index !== -1) {
    heroProfiles.splice(index, 1);
  }
  
  console.log(heroProfiles);
