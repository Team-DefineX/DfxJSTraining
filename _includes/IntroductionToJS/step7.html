The let statement in JavaScript is used to declare variables that are scoped to the nearest enclosing block. Another term for this is called block scope variable.


let statement was introduced in JavaScript 1.7, providing JavaScript with more stable functionality.


JavaScript will automatically assign an undefined value to the variables if no value is given to them at the time of declaration.


Variables cannot be used prior to their declaration; otherwise, an error occurs. 

    
    let variable_name = value;
Parameter
  • let: JavaScript reserved keyword.
  • constant_name: Name of the variable.
  • value: Optional, Initial value assign at the time of declaration.

You can declare one or more variables in a single statement separated by a comma. 

    
    let var1 = val1, var2 = val2, ..., varN = valN;

let in functional scope: In the following let statement example, variable scope is used within function. 

    
    function callme() {
        let printText = "Team DefineX";
        console.log("inside: "+ printText); // inside: Team DefineX
    }

    callme();
    console.log("outside: "+ printText); // Uncaught ReferenceError: printText is not defined"

let in loops: let variable can be seen in the for() loop block. 

    
    function callme() {
        for( let i = 1; i < 5; i++ ) {
            console.log(i);                    // 1, 2, 3, 4
        };
        
        console.log("Outside for block:", i);  // i not defined   
    }

    callme();

let variable is visible in the if condition block. 

    
    if (true) {
        let i;
        for( i = 1; i < 5; i++ ) {
            console.log(i);  // 1, 2, 3, 4
        };
    }

    console.log(i);          // Uncaught ReferenceError: i is not defined"

let in global scope: let keyword can be utilized to define the variable globally. Once you define it globally, you can access it both inside and outside of the function. 

    
    let i = 1;
    console.log("start----", i);                    // start---- 1
    
    function callme() {
        for( i = 1; i < 5; i++ ) {
            console.log("looping time----",i);      // looping time---- 1, 2, 3, 4, 5 
        };
        console.log("function end----", i);         // function end---- 5
    }  
 
    callme();
    console.log("end----", i);                    // end---- 5
